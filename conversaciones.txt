```python
# procesar.py

# Importamos las bibliotecas necesarias
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer

# Inicializamos las bibliotecas
nltk.download('punkt')
nltk.download('wordnet')
nltk.download('stopwords')

# Creamos un lematizador de palabras
lemmatizer = WordNetLemmatizer()

# Creamos una lista de stopwords en inglés
stop_words = set(stopwords.words('english'))

def procesar_texto(texto):
    # Tokenizamos el texto en palabras
    palabras = word_tokenize(texto)

    # Eliminamos stopwords y lematizamos las palabras
    palabras_procesadas = [lemmatizer.lemmatize(palabra.lower()) for palabra in palabras if palabra.isalpha() and palabra.lower() not in stop_words]

    # Volvemos a unir las palabras procesadas en un texto
    texto_procesado = ' '.join(palabras_procesadas)

    return texto_procesado

# Ejemplo de uso
texto = "Este es un ejemplo de texto que necesito procesar."
texto_procesado = procesar_texto(texto)
print(texto_procesado)
```
